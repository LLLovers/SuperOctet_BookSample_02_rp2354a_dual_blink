cmake_minimum_required(VERSION 3.25)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to Pico SDK 2.3.1")
endif()
set(PICO_BOARD hardware_config CACHE STRING "RP2354A board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Cortex-M33 secure platform")
if(NOT PICO_BOARD STREQUAL "hardware_config" OR NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
    message(FATAL_ERROR "This sample requires Aether RP2354A / rp2350-arm-s")
endif()
set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(BookRP2354ABlink LANGUAGES C CXX ASM)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING VERSION_EQUAL "2.3.1")
    message(FATAL_ERROR "Pico SDK 2.3.1 is required")
endif()
if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_VERSION VERSION_LESS 12)
    message(FATAL_ERROR "Use Arm GNU GCC 12 or newer")
endif()
add_executable(rp2354a_dual_blink main.cpp)
target_compile_options(rp2354a_dual_blink PRIVATE -Os -g3 -Wall -Wextra)
target_link_libraries(rp2354a_dual_blink PRIVATE pico_stdlib pico_multicore)
pico_enable_stdio_usb(rp2354a_dual_blink 0)
pico_enable_stdio_uart(rp2354a_dual_blink 0)
pico_add_extra_outputs(rp2354a_dual_blink)
